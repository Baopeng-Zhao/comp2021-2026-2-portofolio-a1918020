using System;
using System.Collections.Generic;
using System.Linq;

// Build the Horse graph.
Graph graph = new Graph();
Node Start = new Node("Start");
Node A = new Node("A");
Node B = new Node("B (BEES)");
Node C = new Node("C");
Node D = new Node("D (BEES)");
Node E = new Node("E (BEES)");
Node F = new Node("F");
Node G = new Node("G");
Node H = new Node("H");
Node I = new Node("I (BEES)");
Node J = new Node("J (BEES)");
Node K = new Node("K");
Node L = new Node("L");
Node M = new Node("M");
Node N = new Node("N");
Node End = new Node("End");

graph.AddEdge(Start, A, 1);
graph.AddEdge(Start, B, 5);
graph.AddEdge(Start, D, 5);

graph.AddEdge(A, C, 1);
graph.AddEdge(B, E, 5);

graph.AddEdge(D, C, 5);
graph.AddEdge(D, E, 5);
graph.AddEdge(D, G, 5);

graph.AddEdge(E, H, 5);
graph.AddEdge(C, F, 1);

graph.AddEdge(G, F, 1);
graph.AddEdge(G, H, 1);
graph.AddEdge(G, J, 1);

graph.AddEdge(F, I, 5);
graph.AddEdge(H, K, 1);

graph.AddEdge(J, I, 5);
graph.AddEdge(J, K, 5);
graph.AddEdge(J, M, 5);

graph.AddEdge(I, L, 5);
graph.AddEdge(K, N, 1);

graph.AddEdge(M, L, 1);
graph.AddEdge(M, N, 1);

graph.AddEdge(L, End, 1);

// Prepare Dijkstra's data structures.
Node target = End;
Dictionary<Node, int> distanceList = new Dictionary<Node, int>();
Dictionary<Node, Node?> parents = new Dictionary<Node, Node?>();
PriorityQueue<Node, int> priorityQueue = new PriorityQueue<Node, int>();

foreach (Node node in graph.Nodes)
{
    distanceList[node] = int.MaxValue;
}

distanceList[Start] = 0;
parents[Start] = null;
priorityQueue.Enqueue(Start, 0);

// Find the cheapest route.
while (priorityQueue.TryDequeue(out Node? current, out int queuedDistance))
{
    if (current == null) continue;

    // Ignore old queue entries if a shorter path was found.
    if (queuedDistance != distanceList[current]) continue;

    if (current == target) break;

    foreach (KeyValuePair<Node, int> neighbour in current.Edges)
    {
        Node next = neighbour.Key;
        int weight = neighbour.Value;

        // Guard against integer overflow.
        if (distanceList[current] > int.MaxValue - weight) continue;

        int newDistance = distanceList[current] + weight;
        if (newDistance >= distanceList[next]) continue;

        distanceList[next] = newDistance;
        parents[next] = current;
        priorityQueue.Enqueue(next, newDistance);
    }
}

Console.WriteLine($"Horse graph: {graph.Nodes.Count} nodes, {graph.Nodes.Sum(n => n.Edges.Count) / 2} edges");

if (distanceList[target] == int.MaxValue)
{
    Console.WriteLine("No path found from Start to End.");
}
else
{
    List<string> path = new List<string>();
    Node? node = target;

    // Follow parent links from End back to Start.
    while (node != null)
    {
        path.Add(node.Data);
        node = parents[node];
    }

    path.Reverse();
    Console.WriteLine("Shortest path: " + string.Join(" -> ", path));
    Console.WriteLine("Total distance: " + distanceList[target]);
}
